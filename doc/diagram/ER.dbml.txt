Project FreelancerHiringPlatform {
  database_type: "MySQL"
  Note: "Freelancer Hiring Platform - Client, Freelancer and Admin"
}

Table users {
  id bigint [pk, increment]
  name varchar(100) [not null]
  email varchar(150) [not null, unique]
  password varchar(255) [not null]
  role varchar(20) [not null, note: 'CLIENT, FREELANCER, ADMIN']
  is_active boolean [not null, default: true]
  created_at datetime [not null]
  updated_at datetime
}

Table freelancer_profiles {
  id bigint [pk, increment]
  user_id bigint [not null, unique]
  bio text
  experience_years int
  portfolio_url varchar(500)
  location varchar(150)
  created_at datetime
  updated_at datetime
}

Table skills {
  id bigint [pk, increment]
  name varchar(100) [not null, unique]
}

Table freelancer_skills {
  id bigint [pk, increment]
  freelancer_id bigint [not null]
  skill_id bigint [not null]
  
  Indexes {
    (freelancer_id, skill_id) [unique]
  }
}

Table projects {
  id bigint [pk, increment]
  client_id bigint [not null]
  title varchar(200) [not null]
  description text [not null]
  budget decimal(12,2) [not null]
  deadline date [not null]
  status varchar(30) [not null, note: 'POSTED, HIRED, IN_PROGRESS, COMPLETED']
  created_at datetime [not null]
  updated_at datetime
}

Table proposals {
  id bigint [pk, increment]
  project_id bigint [not null]
  freelancer_id bigint [not null]
  bid_amount decimal(12,2) [not null]
  estimated_days int [not null]
  cover_letter text
  status varchar(30) [not null, note: 'PENDING, ACCEPTED, REJECTED, WITHDRAWN']
  submitted_at datetime [not null]
  updated_at datetime
}

Table contracts {
  id bigint [pk, increment]
  project_id bigint [not null, unique]
  proposal_id bigint [not null, unique]
  client_id bigint [not null]
  freelancer_id bigint [not null]
  agreed_amount decimal(12,2) [not null]
  start_date date
  end_date date
  status varchar(30) [not null, note: 'ACTIVE, COMPLETED, CANCELLED']
  created_at datetime [not null]
  updated_at datetime
}

Table reviews {
  id bigint [pk, increment]
  contract_id bigint [not null]
  reviewer_id bigint [not null]
  reviewee_id bigint [not null]
  rating int [not null, note: '1 to 5']
  comment text
  created_at datetime [not null]
}

Table notifications {
  id bigint [pk, increment]
  user_id bigint [not null]
  title varchar(200) [not null]
  message text [not null]
  is_read boolean [not null, default: false]
  created_at datetime [not null]
}

Table reports {
  id bigint [pk, increment]
  reporter_id bigint [not null]
  reported_user_id bigint
  project_id bigint
  reason text [not null]
  status varchar(30) [not null, note: 'PENDING, REVIEWED, RESOLVED, REJECTED']
  created_at datetime [not null]
  updated_at datetime
}


// ====================
// USER RELATIONSHIPS
// ====================

Ref: users.id - freelancer_profiles.user_id

Ref: users.id < projects.client_id

Ref: users.id < proposals.freelancer_id

Ref: users.id < contracts.client_id

Ref: users.id < contracts.freelancer_id

Ref: users.id < reviews.reviewer_id

Ref: users.id < reviews.reviewee_id

Ref: users.id < notifications.user_id

Ref: users.id < reports.reporter_id

Ref: users.id < reports.reported_user_id


// ====================
// FREELANCER RELATIONSHIPS
// ====================

Ref: freelancer_profiles.id < freelancer_skills.freelancer_id

Ref: skills.id < freelancer_skills.skill_id


// ====================
// PROJECT RELATIONSHIPS
// ====================

Ref: projects.id < proposals.project_id

Ref: projects.id - contracts.project_id

Ref: proposals.id - contracts.proposal_id


// ====================
// REVIEW & REPORT
// ====================

Ref: contracts.id < reviews.contract_id

Ref: projects.id < reports.project_id
